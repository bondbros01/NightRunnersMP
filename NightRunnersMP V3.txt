    /// <summary>
    /// Exact 58-byte blittable network telemetry packet layout.
    /// </summary>
    [StructLayout(LayoutKind.Sequential, Pack = 1)]
    public struct CarNetworkPacket
    {
        public byte PacketHeader;
        public uint SequenceId;
        public float Timestamp;
        public float PositionX;
        public float PositionY;
        public float PositionZ;
        public float RotationX;
        public float RotationY;
        public float RotationZ;
        public float RotationW;
        public float VelocityX;
        public float VelocityY;
        public float VelocityZ;
        public float SteerAngle;
        public float Revs;
        public float BrakeInput;
        public byte Skidding;
    }

    /// <summary>
    /// Tracks active remote ghost players and manages vehicle instantiation/destruction.
    /// </summary>
    public static class RemotePlayerManager
    {
        private static readonly Dictionary<ulong, RemoteCarInterpolator> RemotePlayers = new Dictionary<ulong, RemoteCarInterpolator>();

        public static void ProcessIncomingPacket(CSteamID senderSteamId, in CarNetworkPacket packet, GameObject localVehicleTemplate)
        {
            ulong steamId64 = senderSteamId.m_SteamID;

            if (!RemotePlayers.TryGetValue(steamId64, out RemoteCarInterpolator interpolator) || interpolator == null)
            {
                if (localVehicleTemplate == null) return;

                interpolator = SpawnGhostVehicle(senderSteamId, localVehicleTemplate, packet);
                if (interpolator != null)
                {
                    RemotePlayers[steamId64] = interpolator;
                }
                return;
            }

            interpolator.UpdateTargetState(packet);
        }

        private static RemoteCarInterpolator SpawnGhostVehicle(CSteamID senderSteamId, GameObject localVehicleTemplate, in CarNetworkPacket initialPacket)
        {
            NightrunnersMPPlugin.Log.LogInfo("Spawning new Ghost Vehicle for remote Steam ID: " + senderSteamId.m_SteamID);

            Vector3 spawnPosition = new Vector3(initialPacket.PositionX, initialPacket.PositionY, initialPacket.PositionZ);
            Quaternion spawnRotation = new Quaternion(initialPacket.RotationX, initialPacket.RotationY, initialPacket.RotationZ, initialPacket.RotationW);

            GameObject ghostCar = UnityEngine.Object.Instantiate(localVehicleTemplate, spawnPosition, spawnRotation);
            ghostCar.name = "GhostVehicle_" + senderSteamId.m_SteamID;
            ghostCar.tag = "Untagged";

            // Strip audio components string-safely to bypass missing references
            var allComponents = ghostCar.GetComponentsInChildren(Il2CppType.Of<Component>());
            for (int i = 0; i < allComponents.Length; i++)
            {
                Component comp = allComponents[i];
                if (comp != null && (comp.GetIl2CppType().Name.Contains("AudioListener") || comp.GetIl2CppType().Name.Contains("CarAudio")))
                {
                    UnityEngine.Object.Destroy(comp);
                }
            }

            // Neutralize local input control scripts string-safely
            Component userControl = ghostCar.GetComponent("CarUserControl");
            if (userControl != null) UnityEngine.Object.Destroy(userControl);

            CarController carController = ghostCar.GetComponent<CarController>();
            Transform frontLeftWheelTransform = null;
            Transform frontRightWheelTransform = null;

            if (carController != null)
            {
                carController.enabled = false;
                UnityEngine.Object.Destroy(carController);
            }

            // Fallback wheel mesh discovery crawling the visual asset hierarchy
            Transform[] allChildren = ghostCar.GetComponentsInChildren<Transform>();
            for (int i = 0; i < allChildren.Length; i++)
            {
                Transform t = allChildren[i];
                string lowerName = t.name.ToLower();
                if (frontLeftWheelTransform == null && (lowerName.Contains("wheel_fl") || lowerName.Contains("wheel_l_front") || lowerName.Contains("fl_wheel")))
                {
                    frontLeftWheelTransform = t;
                }
                else if (frontRightWheelTransform == null && (lowerName.Contains("wheel_fr") || lowerName.Contains("wheel_r_front") || lowerName.Contains("fr_wheel")))
                {
                    frontRightWheelTransform = t;
                }
            }

            Rigidbody rb = ghostCar.GetComponent<Rigidbody>();
            if (rb != null)
            {
                rb.isKinematic = true;
                rb.useGravity = false;
                rb.velocity = Vector3.zero;
                rb.angularVelocity = Vector3.zero;
            }

            // Disable physical wheel colliders safely via a string-based behaviour loop
            for (int i = 0; i < allComponents.Length; i++)
            {
                Component comp = allComponents[i];
                if (comp != null && comp.GetIl2CppType().Name.Contains("WheelCollider"))
                {
                    Behaviour behaviourComp = comp.TryCast<Behaviour>();
                    if (behaviourComp != null) behaviourComp.enabled = false;
                }
            }

            // Setup dedicated 3D AudioSource for engine noise
            AudioSource engineAudio = ghostCar.GetComponentInChildren<AudioSource>();
            if (engineAudio == null)
            {
                engineAudio = ghostCar.AddComponent(Il2CppType.Of<AudioSource>()).TryCast<AudioSource>();
            }
            if (engineAudio != null)
            {
                engineAudio.spatialBlend = 1.0f;
                engineAudio.minDistance = 3.0f;
                engineAudio.maxDistance = 60.0f;
                engineAudio.loop = true;
                if (engineAudio.clip != null && !engineAudio.isPlaying)
                {
                    engineAudio.Play();
                }
            }

            // Attach the state interpolation driver via standard IL2CPP formatting
            RemoteCarInterpolator interpolator = ghostCar.AddComponent(Il2CppType.Of<RemoteCarInterpolator>()).TryCast<RemoteCarInterpolator>();
            if (interpolator != null)
            {
                interpolator.Initialize(frontLeftWheelTransform, frontRightWheelTransform, engineAudio);
                interpolator.UpdateTargetState(initialPacket);
            }

            return interpolator;
        }

        public static void RemovePlayer(ulong steamId64)
        {
            if (RemotePlayers.TryGetValue(steamId64, out RemoteCarInterpolator interpolator))
            {
                if (interpolator != null && interpolator.gameObject != null)
                {
                    UnityEngine.Object.Destroy(interpolator.gameObject);
                }
                RemotePlayers.Remove(steamId64);
                NightrunnersMPPlugin.Log.LogInfo("Removed remote player ghost vehicle: " + steamId64);
            }
        }

        public static void ClearAllPlayers()
        {
            foreach (var kvp in RemotePlayers)
            {
                if (kvp.Value != null && kvp.Value.gameObject != null)
                {
                    UnityEngine.Object.Destroy(kvp.Value.gameObject);
                }
            }
            RemotePlayers.Clear();
            NightrunnersMPPlugin.Log.LogInfo("Cleaned up all remote player vehicles.");
        }
    }

    /// <summary>
    /// Smooths spatial position, dead reckoning vectors, wheel mesh rotation, and engine audio.
    /// </summary>
    public class RemoteCarInterpolator : MonoBehaviour
    {
        public RemoteCarInterpolator(System.IntPtr ptr) : base(ptr) { }

        private Vector3 _targetPosition;
        private Quaternion _targetRotation;
        private Vector3 _targetVelocity;
        private float _targetSteerAngle;
        private float _currentSteerAngle;
        private float _targetRevs;
        private float _currentRevs;

        private Transform _frontLeftWheel;
        private Transform _frontRightWheel;
        private Quaternion _flBaseRotation;
        private Quaternion _frBaseRotation;
        private AudioSource _engineAudio;
        private uint _lastSequence;

        private const float PositionLerpFactor = 15.0f;
        private const float RotationSlerpFactor = 15.0f;
        private const float TeleportThreshold = 25.0f;

        public void Initialize(Transform frontLeft, Transform frontRight, AudioSource audio)
        {
            _frontLeftWheel = frontLeft;
            _frontRightWheel = frontRight;
            _engineAudio = audio;

            if (_frontLeftWheel != null) _flBaseRotation = _frontLeftWheel.localRotation;
            if (_frontRightWheel != null) _frBaseRotation = _frontRightWheel.localRotation;

            _targetPosition = transform.position;
            _targetRotation = transform.rotation;
        }

        public void UpdateTargetState(in CarNetworkPacket packet)
        {
            if (packet.SequenceId < _lastSequence && (_lastSequence - packet.SequenceId) < 1000) return;
            _lastSequence = packet.SequenceId;

            _targetPosition = new Vector3(packet.PositionX, packet.PositionY, packet.PositionZ);
            _targetRotation = new Quaternion(packet.RotationX, packet.RotationY, packet.RotationZ, packet.RotationW);
            _targetVelocity = new Vector3(packet.VelocityX, packet.VelocityY, packet.VelocityZ);
            _targetSteerAngle = packet.SteerAngle;
            _targetRevs = packet.Revs;

            if (Vector3.Distance(transform.position, _targetPosition) > TeleportThreshold)
            {
                transform.position = _targetPosition;
                transform.rotation = _targetRotation;
            }
        }

        private void Update()
        {
            float dt = Time.deltaTime;

            Vector3 extrapolatedPosition = _targetPosition + (_targetVelocity * dt);

            transform.position = Vector3.Lerp(transform.position, extrapolatedPosition, dt * PositionLerpFactor);
            transform.rotation = Quaternion.Slerp(transform.rotation, _targetRotation, dt * RotationSlerpFactor);

            _currentSteerAngle = Mathf.Lerp(_currentSteerAngle, _targetSteerAngle, dt * 15.0f);
            _currentRevs = Mathf.Lerp(_currentRevs, _targetRevs, dt * 8.0f);

            Quaternion steerOffset = Quaternion.Euler(0.0f, _currentSteerAngle, 0.0f);
            if (_frontLeftWheel != null)
            {
                _frontLeftWheel.localRotation = _flBaseRotation * steerOffset;
            }
            if (_frontRightWheel != null)
            {
                _frontRightWheel.localRotation = _frBaseRotation * steerOffset;
            }

            if (_engineAudio != null)
            {
                float clampedRevs = Mathf.Clamp01(_currentRevs);
                _engineAudio.pitch = Mathf.Lerp(0.75f, 2.2f, clampedRevs);
            }
        }
    }

    /// <summary>
    /// Managed Steamworks Matchmaking, Lobby Management, and P2P routing subsystem.
    /// </summary>
    public static class SteamNetworkManager
    {
        // Tracking handles for Matchmaking and GUI feedback strings
        public static CSteamID CurrentLobby = CSteamID.Nil;
        public static string StatusMessage = "Ready to Host/Join.";

        public static readonly List<CSteamID> DiscoveredLobbies = new List<CSteamID>();

        // Steamworks Callbacks and CallResults
        private static CallResult<LobbyCreated_t> _lobbyCreatedCallResult;
        private static CallResult<LobbyEnter_t> _lobbyEnteredCallResult;
        private static CallResult<LobbyMatchList_t> _lobbyMatchListCallResult;
        private static Callback<LobbyChatUpdate_t> _lobbyChatUpdateCallback;

        public static void Initialize()
        {
            if (!SteamAPI.IsSteamRunning()) return;

            _lobbyCreatedCallResult = CallResult<LobbyCreated_t>.Create(OnLobbyCreated);
            _lobbyEnteredCallResult = CallResult<LobbyEnter_t>.Create(OnLobbyEntered);
            _lobbyMatchListCallResult = CallResult<LobbyMatchList_t>.Create(OnLobbyMatchList);
            _lobbyChatUpdateCallback = Callback<LobbyChatUpdate_t>.Create(OnLobbyChatUpdate);
        }

        public static void OnLobbyCreated(LobbyCreated_t callback)
        {
            if (callback.m_eResult == EResult.k_EResultOK)
            {
                CurrentLobby = new CSteamID(callback.m_ulSteamIDLobby);
                StatusMessage = "Lobby Created! Join code copied.";
                GUIUtility.systemCopyBuffer = callback.m_ulSteamIDLobby.ToString();
            }
            else
            {
                StatusMessage = "Lobby creation failed: " + callback.m_eResult;
            }
        }

        public static void OnLobbyEnter(LobbyEnter_t callback)
        {
            CurrentLobby = new CSteamID(callback.m_ulSteamIDLobby);
            StatusMessage = "Session Connected! Data pipeline open.";
        }


        public static void RefreshLobbyList()
        {
            if (!SteamAPI.IsSteamRunning()) return;
            StatusMessage = "Searching for lobbies...";
            SteamMatchmaking.AddRequestLobbyListStringFilter("game", "NightrunnersMP", ELobbyComparison.k_ELobbyComparisonEqual);
            SteamAPICall_t handle = SteamMatchmaking.RequestLobbyList();
            _lobbyMatchListCallResult.Set(handle);
        }

        public static void LeaveLobby()
        {
            if (CurrentLobby.IsValid() && CurrentLobby != CSteamID.Nil)
            {
                SteamMatchmaking.LeaveLobby(CurrentLobby);
                CurrentLobby = CSteamID.Nil;
                RemotePlayerManager.ClearAllPlayers();
                StatusMessage = "Left lobby.";
                NightrunnersMPPlugin.Log.LogInfo("Disconnected from lobby.");
            }
        }

        private static void OnLobbyCreated(LobbyCreated_t callback, bool bIOFailure)
        {
            if (bIOFailure || callback.m_eResult != EResult.k_EResultOK)
            {
                StatusMessage = "Lobby creation failed: " + callback.m_eResult;
                NightrunnersMPPlugin.Log.LogError(StatusMessage);
                return;
            }

            CurrentLobby = new CSteamID(callback.m_ulSteamIDLobby);
            SteamMatchmaking.SetLobbyData(CurrentLobby, "game", "NightrunnersMP");
            SteamMatchmaking.SetLobbyData(CurrentLobby, "name", SteamFriends.GetPersonaName() + "'s Session");
            StatusMessage = "Lobby Created successfully!";
            NightrunnersMPPlugin.Log.LogInfo("Lobby Created. ID: " + CurrentLobby.m_SteamID);
        }

        private static void OnLobbyEntered(LobbyEnter_t callback, bool bIOFailure)
        {
            if (bIOFailure || callback.m_EChatRoomEnterResponse != (uint)EChatRoomEnterResponse.k_EChatRoomEnterResponseSuccess)
            {
                StatusMessage = "Failed to enter lobby.";
                NightrunnersMPPlugin.Log.LogError(StatusMessage);
                return;
            }

            CurrentLobby = new CSteamID(callback.m_ulSteamIDLobby);
            StatusMessage = "Connected to lobby.";
            NightrunnersMPPlugin.Log.LogInfo("Joined lobby: " + CurrentLobby.m_SteamID);
        }

        private static void OnLobbyMatchList(LobbyMatchList_t callback, bool bIOFailure)
        {
            DiscoveredLobbies.Clear();
            if (bIOFailure)
            {
                StatusMessage = "Failed to query lobby list.";
                return;
            }

            for (int i = 0; i < (int)callback.m_nLobbiesMatching; i++)
            {
                CSteamID lobby = SteamMatchmaking.GetLobbyByIndex(i);
                DiscoveredLobbies.Add(lobby);
            }
            StatusMessage = "Found " + DiscoveredLobbies.Count + " active lobbies.";
        }

        private static void OnLobbyChatUpdate(LobbyChatUpdate_t callback)
        {
            EChatMemberStateChange stateChange = (EChatMemberStateChange)callback.m_rgfChatMemberStateChange;
            if (stateChange == EChatMemberStateChange.k_EChatMemberStateChangeLeft ||
                stateChange == EChatMemberStateChange.k_EChatMemberStateChangeDisconnected ||
                stateChange == EChatMemberStateChange.k_EChatMemberStateChangeKicked ||
                stateChange == EChatMemberStateChange.k_EChatMemberStateChangeBanned)
            {
                RemotePlayerManager.RemovePlayer(callback.m_ulSteamIDUserChanged);
            }
        }

        public static void DispatchToSteamP2P(byte[] payload, int size)
        {
            if (!SteamAPI.IsSteamRunning()) return;
            if (!CurrentLobby.IsValid() || CurrentLobby == CSteamID.Nil) return;

            int memberCount = SteamMatchmaking.GetNumLobbyMembers(CurrentLobby);
            CSteamID localSteamId = SteamUser.GetSteamID();

            for (int i = 0; i < memberCount; i++)
            {
                CSteamID memberId = SteamMatchmaking.GetLobbyMemberByIndex(CurrentLobby, i);
                if (memberId == localSteamId || !memberId.IsValid()) continue;

                SteamNetworking.SendP2PPacket(memberId, payload, (uint)size, EP2PSend.k_EP2PSendUnreliableNoDelay, 0);
            }
        }

        public static unsafe void ReceiveIncomingPackets(byte[] receiveBuffer, int packetSizeTarget, GameObject localVehicleTemplate)
        {
            if (!SteamAPI.Init()) return;

            uint packetSize;
            while (SteamNetworking.IsP2PPacketAvailable(out packetSize, 0))
            {
                if (packetSize == (uint)packetSizeTarget)
                {
                    uint bytesRead;
                    CSteamID remoteSteamId;

                    if (SteamNetworking.ReadP2PPacket(receiveBuffer, packetSize, out bytesRead, out remoteSteamId, 0))
                    {
                        fixed (byte* ptr = receiveBuffer)
                        {
                            CarNetworkPacket packet = *(CarNetworkPacket*)ptr;
                            if (packet.PacketHeader == 0x01)
                            {
                                RemotePlayerManager.ProcessIncomingPacket(remoteSteamId, in packet, localVehicleTemplate);
                            }
                        }
                    }
                }
                else
                {
                    byte[] discardBuffer = new byte[packetSize];
                    SteamNetworking.ReadP2PPacket(discardBuffer, packetSize, out _, out _, 0);
                }
            }
        }
    }

    /// <summary>
    /// Central ticker managing local telemetry transmission, packet reception, and legacy GUI menu rendering.
    /// </summary>
    public class NetworkVehicleTicker : MonoBehaviour
    {
        public NetworkVehicleTicker(System.IntPtr ptr) : base(ptr) { }

        private CarController _localCarController;
        private Rigidbody _localRigidbody;
        private uint _packetSequenceCounter;

        private static readonly int PacketSize = Marshal.SizeOf<CarNetworkPacket>();
        private readonly byte[] _sendBuffer = new byte[PacketSize];
        private readonly byte[] _receiveBuffer = new byte[PacketSize];

        private float _nextScanTime;
        private const float ScanInterval = 1.0f;

        private float _nextTickTime;
        private const float NetworkSendRate = 1.0f / 30.0f;

        // GUI States
        private bool _showMenu = false;
        private string _directJoinInput = "";
        private Vector2 _lobbyScrollPos;
        private Vector2 _playerScrollPos;

        private void Update()
        {
            // Pump Steamworks async event callbacks
            if (SteamAPI.IsSteamRunning())
            {
                if (SteamAPI.Init())
                {
                    SteamAPI.RunCallbacks();
                }
            }

            // Toggle Graphical Matchmaking Menu with F1
            if (Input.GetKeyDown(KeyCode.F1))
            {
                _showMenu = !_showMenu;
                Cursor.visible = _showMenu;
                Cursor.lockState = _showMenu ? CursorLockMode.None : CursorLockMode.Locked;
            }

            // 1. Maintain local player vehicle discovery
            if (_localCarController == null || _localRigidbody == null)
            {
                if (Time.unscaledTime >= _nextScanTime)
                {
                    _nextScanTime = Time.unscaledTime + ScanInterval;
                    LocateLocalPlayerVehicle();
                }
            }

            // 2. Poll & Process incoming remote packets
            GameObject template = _localCarController != null ? _localCarController.gameObject : null;
            SteamNetworkManager.ReceiveIncomingPackets(_receiveBuffer, PacketSize, template);

            // 3. Broadcast local vehicle telemetry
            if (_localCarController != null && _localRigidbody != null && Time.unscaledTime >= _nextTickTime)
            {
                _nextTickTime = Time.unscaledTime + NetworkSendRate;
                BroadcastVehicleState();
            }
        }

        private void LocateLocalPlayerVehicle()
        {
            CarController[] allCars = UnityEngine.Object.FindObjectsOfType<CarController>();
            if (allCars == null || allCars.Length == 0) return;

            for (int i = 0; i < allCars.Length; i++)
            {
                CarController car = allCars[i];
                if (car == null || car.gameObject.name.StartsWith("GhostVehicle_")) continue;

                Component userControl = car.GetComponent("CarUserControl");
                bool hasPlayerTag = car.CompareTag("Player") || car.transform.root.CompareTag("Player");
                bool isCameraTarget = Camera.main != null && Camera.main.transform.parent != null &&
                                     Camera.main.transform.parent.root == car.transform.root;

                if (userControl != null || hasPlayerTag || isCameraTarget)
                {
                    _localCarController = car;
                    _localRigidbody = car.GetComponent(Il2CppType.Of<Rigidbody>()).TryCast<Rigidbody>();
                    NightrunnersMPPlugin.Log.LogInfo("Resolved local vehicle instance: " + car.gameObject.name);
                    break;
                }
            }
        }

        private unsafe void BroadcastVehicleState()
        {
            Vector3 pos = _localRigidbody.position;
            Quaternion rot = _localRigidbody.rotation;
            Vector3 vel = _localRigidbody.velocity;

            CarNetworkPacket packet = new CarNetworkPacket
            {
                PacketHeader = 0x01,
                SequenceId = ++_packetSequenceCounter,
                Timestamp = Time.unscaledTime,

                PositionX = pos.x,
                PositionY = pos.y,
                PositionZ = pos.z,

                RotationX = rot.x,
                RotationY = rot.y,
                RotationZ = rot.z,
                RotationW = rot.w,

                VelocityX = vel.x,
                VelocityY = vel.y,
                VelocityZ = vel.z,

                SteerAngle = _localCarController.CurrentSteerAngle,
                Revs = _localCarController.Revs,
                BrakeInput = _localCarController.BrakeInput,
                Skidding = (byte)(_localCarController.Skidding ? 1 : 0)
            };

            fixed (byte* bufferPtr = _sendBuffer)
            {
                *(CarNetworkPacket*)bufferPtr = packet;
            }

            SteamNetworkManager.DispatchToSteamP2P(_sendBuffer, PacketSize);
        }

        private void OnGUI()
        {
            // Discreet Watermark in upper-left corner
            string lobbyStateText = SteamNetworkManager.CurrentLobby.IsValid() && SteamNetworkManager.CurrentLobby != CSteamID.Nil
                ? "<color=#55FF55>IN LOBBY</color>"
                : "<color=#FFAA00>OFFLINE</color>";

            GUI.Label(new Rect(10, 10, 350, 25), $"<b>Nightrunners P2P</b> | {lobbyStateText} | Press <b>[F1]</b> for Menu");

            if (!_showMenu) return;

            // Render Matchmaking Menu using delegate-free GUILayout Area (100% safe in IL2CPP)
            GUILayout.BeginArea(new Rect(20, 40, 420, 520), GUI.skin.box);
            GUILayout.Label("<b><size=16>NIGHTRUNNERS MULTIPLAYER</size></b>");
            GUILayout.Label("<color=#CCCCCC>Status: " + SteamNetworkManager.StatusMessage + "</color>");
            GUILayout.Space(6);

            bool inLobby = SteamNetworkManager.CurrentLobby.IsValid() && SteamNetworkManager.CurrentLobby != CSteamID.Nil;

            if (inLobby)
            {
                // LOBBY VIEW
                GUILayout.Label($"<b>Active Lobby ID:</b> {SteamNetworkManager.CurrentLobby.m_SteamID}");
                if (GUILayout.Button("Copy Lobby ID to Clipboard"))
                {
                    GUIUtility.systemCopyBuffer = SteamNetworkManager.CurrentLobby.m_SteamID.ToString();
                    SteamNetworkManager.StatusMessage = "Lobby ID copied!";
                }

                GUILayout.Space(8);
                int memberCount = SteamMatchmaking.GetNumLobbyMembers(SteamNetworkManager.CurrentLobby);
                CSteamID lobbyOwner = SteamMatchmaking.GetLobbyOwner(SteamNetworkManager.CurrentLobby);
                CSteamID localSteamId = SteamUser.GetSteamID();

                GUILayout.Label($"<b>Players in Session ({memberCount}/4):</b>");
                _playerScrollPos = GUILayout.BeginScrollView(_playerScrollPos, GUILayout.Height(180));
                for (int i = 0; i < memberCount; i++)
                {
                    CSteamID member = SteamMatchmaking.GetLobbyMemberByIndex(SteamNetworkManager.CurrentLobby, i);
                    string memberName = SteamFriends.GetFriendPersonaName(member);
                    string tag = "";
                    if (member == lobbyOwner) tag += " <color=#FFDD44>[HOST]</color>";
                    if (member == localSteamId) tag += " <color=#66CCFF>[YOU]</color>";

                    GUILayout.Label($" • {memberName}{tag}");
                }
                GUILayout.EndScrollView();

                GUILayout.Space(10);
                if (GUILayout.Button("Invite Friends via Steam"))
                {
                    SteamFriends.ActivateGameOverlay("LobbyInvite");
                }

                if (GUILayout.Button("Leave Lobby"))
                {
                    SteamNetworkManager.LeaveLobby();
                }
            }
            else
            {
                // MATCHMAKING & SEARCH VIEW
                GUILayout.Label("<b>Host Session:</b>");
                GUILayout.BeginHorizontal();
                if (GUILayout.Button("Create Public Lobby"))
                {
                    SteamMatchmaking.CreateLobby(ELobbyType.k_ELobbyTypePublic, 4);
                }
                if (GUILayout.Button("Create Friends-Only"))
                {
                    SteamMatchmaking.CreateLobby(ELobbyType.k_ELobbyTypeFriendsOnly, 4);
                }
                GUILayout.EndHorizontal();

                GUILayout.Space(8);
                GUILayout.Label("<b>Direct Join via Steam ID:</b>");
                GUILayout.BeginHorizontal();
                _directJoinInput = GUILayout.TextField(_directJoinInput);
                if (GUILayout.Button("Join", GUILayout.Width(70)))
                {
                    if (ulong.TryParse(_directJoinInput.Trim(), out ulong targetLobbyId))
                    {
                        SteamMatchmaking.JoinLobby(new CSteamID(targetLobbyId));
                    }
                    else
                    {
                        SteamNetworkManager.StatusMessage = "Invalid Steam ID!";
                    }
                }
                GUILayout.EndHorizontal();

                GUILayout.Space(8);
                GUILayout.BeginHorizontal();
                GUILayout.Label("<b>Public Sessions Browser:</b>");
                if (GUILayout.Button("Refresh", GUILayout.Width(80)))
                {
                    SteamNetworkManager.RefreshLobbyList();
                }
                GUILayout.EndHorizontal();

                _lobbyScrollPos = GUILayout.BeginScrollView(_lobbyScrollPos, GUILayout.Height(170));
                if (SteamNetworkManager.DiscoveredLobbies.Count == 0)
                {
                    GUILayout.Label("<color=#888888><i>No public lobbies discovered. Click Refresh to scan.</i></color>");
                }
                else
                {
                    for (int i = 0; i < SteamNetworkManager.DiscoveredLobbies.Count; i++)
                    {
                        CSteamID lobby = SteamNetworkManager.DiscoveredLobbies[i];
                        string lobbyName = SteamMatchmaking.GetLobbyData(lobby, "name");
                        if (string.IsNullOrEmpty(lobbyName)) lobbyName = "Nightrunners Session";
                        int count = SteamMatchmaking.GetNumLobbyMembers(lobby);

                        GUILayout.BeginHorizontal();
                        GUILayout.Label($"{lobbyName} ({count}/4)");
                        if (GUILayout.Button("Join", GUILayout.Width(60)))
                        {
                            SteamMatchmaking.JoinLobby(lobby);
                        }
                        GUILayout.EndHorizontal();
                    }
                }
                GUILayout.EndScrollView();
            }

            GUILayout.FlexibleSpace();
            if (GUILayout.Button("Close Menu [F1]"))
            {
                _showMenu = false;
                Cursor.visible = false;
                Cursor.lockState = CursorLockMode.Locked;
            }
            GUILayout.EndArea();
        }
    }
